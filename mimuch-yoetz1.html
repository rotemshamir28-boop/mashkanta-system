export default { async fetch(request, env) { // 1. הגדרת כותרות CORS מקיפות const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type, Authorization, x-requested-with", "Access-Control-Max-Age": "86400", // שמירת התשובה בזיכרון המטמון של הדפדפן ל-24 שעות }; // 2. טיפול מיידי בבקשת Preflight (OPTIONS) // חייב להיות בראש ה-Worker לפני כל פעולה אחרת! if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: corsHeaders, }); } // 3. אישור מתודת POST בלבד if (request.method !== "POST") { return new Response( JSON.stringify({ error: "Method not allowed" }), { status: 405, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } try { const body = await request.json().catch(() => ({})); const images = body.images || []; if (!images || images.length === 0) { return new Response( JSON.stringify({ error: "לא התקבלו תמונות לעיבוד" }), { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } // 4. בדיקת מפתח API מול Cloudflare Environment Variables const apiKey = env.ANTHROPIC_API_KEY; if (!apiKey) { return new Response( JSON.stringify({ error: "חסר מפתח ANTHROPIC_API_KEY בהגדרות ה-Worker" }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } // 5. הכנת התוכן לבקשת ה-AI const content = []; for (const img of images) { if (img.isPdf) { content.push({ type: "document", source: { type: "base64", media_type: img.mime || "application/pdf", data: img.base64 } }); } else { content.push({ type: "image", source: { type: "base64", media_type: img.mime || "image/jpeg", data: img.base64 } }); } } content.push({ type: "text", text: `חלץ את מסלולי המשכנתא מתוך המסמך/התמונות. החזר אך ורק אובייקט JSON תקין (ללא תגיות markdown וללא טקסט חופשי) במבנה הבא: { "tracks": [ { "type": "prime / fixed_linked / fixed_unlinked / var_linked_s / var_unlinked_s / var_linked_l / var_unlinked_l", "remainingPrincipal": מספר_יתרה_בשח, "remainingYears": שנים_שנותרו_כמספר, "rate": אחוז_ריבית_נומינלית_כמספר } ] }` }); // 6. קריאה ל-Anthropic Claude API const response = await fetch("https://api.anthropic.com/v1/messages", { method: "POST", headers: { "x-api-key": apiKey, "anthropic-version": "2023-06-01", "content-type": "application/json" }, body: JSON.stringify({ model: "claude-3-5-sonnet-20241022", max_tokens: 1000, messages: [{ role: "user", content }] }) }); if (!response.ok) { const errText = await response.text(); console.error("Anthropic Error:", errText); return new Response( JSON.stringify({ error: "שגיאה מול מודל ה-AI", details: errText }), { status: 502, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } const aiData = await response.json(); const rawText = aiData.content?.[0]?.text || ""; const cleanJson = rawText.replace(/```json/g, "").replace(/```/g, "").trim(); const parsedData = JSON.parse(cleanJson); return new Response(JSON.stringify(parsedData), { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } }); } catch (err) { console.error("Worker catch error:", err); return new Response( JSON.stringify({ error: err.message || "שגיאה פנימית בשרת" }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } } };
